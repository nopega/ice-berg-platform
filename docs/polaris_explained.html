Polaris ทำงานยังไง และต้องเชื่อมด้วยอะไรบ้าง

Data Platform — Iceberg Lakehouse on EKS · สิงหาคม 2026

1. สถาปัตยกรรม — เกิดอะไรขึ้นตอน query

หัวใจ Polaris ไม่เคยแตะข้อมูลเลย ตอบแค่ 2 อย่าง — "metadata.json ปัจจุบันอยู่ path ไหน" กับ "นี่กุญแจชั่วคราวที่ scope ไว้เฉพาะ prefix นั้น" แล้ว engine ไปคุยกับ S3 เอง

เพราะงั้น pod เดียว 250m CPU รับ warehouse ระดับ TB ได้ — throughput ไม่ได้ไหลผ่านมัน ต่างจาก database ทั่วไปที่ข้อมูลต้องวิ่งผ่านตัว server

2. เชื่อมกันด้วยอะไร — 3 ชั้น

ชั้น 1  Polaris → RDS

อยู่ใน chart/polaris/values.yaml

persistence:
  type: relational-jdbc          # default คือ in-memory ซึ่งพัง
  relationalJdbc:
    secret:
      name: polaris-persistence  # มี jdbcUrl / username / password
จุดที่พลาดแล้วเจ็บ: default ของ chart คือ in-memory ถ้าไม่แก้ พอ pod restart ตัวชี้ตารางหายหมด ไฟล์ Parquet ยังอยู่ใน S3 แต่ไม่มีใครรู้ว่า metadata.json ตัวไหนคือปัจจุบัน = warehouse กลายเป็นขยะ

ชั้น 2  Polaris → S3

ไม่ได้อยู่ใน values.yaml เพราะเป็น property ของ catalog ไม่ใช่ของ server — ผูกตอนรัน 05_create_catalog_prod.sh

"storageConfigInfo": {
  "storageType": "S3",
  "roleArn":     "arn:aws:iam::...:role/data-platform-prod-polaris-storage-role",
  "externalId":  "b058...54c",
  "allowedLocations": ["s3://data-store-prod-warehouse/warehouse"]
}

allowedLocations คือเพดานแข็ง — credential ที่ vend ออกไปเอื้อมนอกลิสต์นี้ไม่ได้ ต่อให้ engine โดนยึด

ตัว pod เองเข้า S3 ผ่าน IRSA แล้ว sts:AssumeRole ต่อไปยัง storage role อีกทอด ทั้งระบบไม่มี AWS access key อยู่เลยสักที่

ชั้น 3  Engine → Polaris (ยังไม่ได้ทำ — ขั้นถัดไป)

# /etc/trino/catalog/data_platform.properties
connector.name=iceberg
iceberg.catalog.type=rest
iceberg.rest-catalog.uri=http://polaris.data-platform.svc.cluster.local:8181/api/catalog
iceberg.rest-catalog.warehouse=data_platform
iceberg.rest-catalog.security=OAUTH2
iceberg.rest-catalog.oauth2.credential=${ENV:POLARIS_CLIENT_ID}:${ENV:POLARIS_CLIENT_SECRET}
iceberg.rest-catalog.vended-credentials-enabled=true

fs.native-s3.enabled=true
s3.region=ap-southeast-1
บรรทัดที่สำคัญที่สุดคือ vended-credentials-enabled=true — บอก Trino ว่าอย่าใช้ IAM ของตัวเอง ให้รับกุญแจจาก Polaris ถ้าไม่ใส่ Trino จะพยายามใช้สิทธิ์ตัวเอง ซึ่ง pod ไม่มี แล้วได้ 403

3. ตอนสร้างตาราง — โค้ดจริง

CREATE SCHEMA data_platform.delivery
WITH (location = 's3://data-store-prod-warehouse/warehouse/delivery');

CREATE TABLE data_platform.delivery.orders (
    order_id      BIGINT,
    merchant_id   BIGINT,
    order_status  VARCHAR,
    total_amount  DECIMAL(12,2),
    created_at    TIMESTAMP(6) WITH TIME ZONE
)
WITH (
    format       = 'PARQUET',
    partitioning = ARRAY['day(created_at)'],
    sorted_by    = ARRAY['merchant_id']
);

hidden partitioning — day(created_at) ทำให้คนเขียน query ไม่ต้องรู้ว่ามี partition column ใส่ WHERE created_at > ... ธรรมดา Iceberg prune partition ให้เอง ต่างจาก Hive ที่ต้องใส่ WHERE dt='2026-08-11' เอง ไม่งั้น full scan

เกิดอะไรขึ้นจริง: ยังไม่มีไฟล์ข้อมูลสักไฟล์ มีแค่ metadata/00000-xxx.metadata.json ใน S3 กับ 1 row ใน Postgres ที่ชี้ไปหามัน

4. ตอนเขียนข้อมูล — Spark (ตัว ETL)

--packages org.apache.iceberg:iceberg-spark-runtime-4.0_2.13:1.10.2,\
           org.apache.iceberg:iceberg-aws-bundle:1.10.2

--conf spark.sql.extensions=org.apache.iceberg.spark.extensions.IcebergSparkSessionExtensions
--conf spark.sql.catalog.data_platform=org.apache.iceberg.spark.SparkCatalog
--conf spark.sql.catalog.data_platform.type=rest
--conf spark.sql.catalog.data_platform.uri=http://polaris.data-platform.svc.cluster.local:8181/api/catalog
--conf spark.sql.catalog.data_platform.warehouse=data_platform
--conf spark.sql.catalog.data_platform.credential=$POLARIS_CLIENT_ID:$POLARIS_CLIENT_SECRET
--conf spark.sql.catalog.data_platform.scope=PRINCIPAL_ROLE:ALL
--conf spark.sql.catalog.data_platform.header.X-Iceberg-Access-Delegation=vended-credentials

บรรทัดสุดท้ายคือฝั่ง Spark ของ vended-credentials-enabled

df = (spark.read
      .option("header", True)
      .csv("s3://data-store-prod-warehouse/raw/orders/*.csv"))

df.createOrReplaceTempView("staged")

spark.sql("""
MERGE INTO data_platform.delivery.orders t
USING staged s
   ON t.order_id = s.order_id
 WHEN MATCHED AND s.order_status <> t.order_status
      THEN UPDATE SET *
 WHEN NOT MATCHED
      THEN INSERT *
""")

ทำไมต้อง MERGE ไม่ใช่ INSERT — MERGE เป็น upsert รันซ้ำกี่รอบผลเท่าเดิม จำเป็นมากเพราะ Airflow retry task ได้ ถ้าใช้ INSERT เฉยๆ retry แล้วข้อมูลซ้ำทันที

เกิดอะไรขึ้นจริง: เขียน Parquet ลง data/created_at_day=2026-08-11/ → เขียน manifest + 00001-xxx.metadata.json → ขอ Polaris สลับ pointer จาก 00000 เป็น 00001 การสลับนั้นคือ commit ถ้าตรงนี้ fail ไฟล์ที่เขียนไปกลายเป็นขยะที่ไม่มีใครมองเห็น ไม่ใช่ข้อมูลครึ่งๆ กลางๆ

5. ตอนดึงข้อมูล — Trino

SELECT merchant_id,
       count(*)          AS orders,
       sum(total_amount) AS revenue
FROM   data_platform.delivery.orders
WHERE  created_at >= TIMESTAMP '2026-08-01 00:00:00 Asia/Bangkok'
GROUP  BY merchant_id
ORDER  BY revenue DESC
LIMIT  20;

ของที่ Hive ทำไม่ได้:

-- time travel: ข้อมูลเมื่อวานเป็นยังไง
SELECT * FROM data_platform.delivery.orders
FOR TIMESTAMP AS OF TIMESTAMP '2026-08-10 10:00:00 Asia/Bangkok';

-- ดูประวัติ commit ทั้งหมด
SELECT * FROM data_platform.delivery."orders$snapshots";

-- เพิ่ม column กลางคัน ไม่ต้อง rewrite ไฟล์เดิม
ALTER TABLE data_platform.delivery.orders ADD COLUMN promo_code VARCHAR;

ALTER TABLE เสร็จทันทีไม่ว่าตารางใหญ่แค่ไหน เพราะแก้แค่ schema ใน metadata.json ไฟล์ Parquet เก่าไม่ถูกแตะเลย — column ใหม่อ่านออกมาเป็น NULL
